def email_valido(email_str: str) -> bool:
    padrao = r'^[\w\.-]+@[\w\.-]+\.\w+$'
    return bool(re.match(padrao, email_str.strip()))

def obter_hora_brasilia():
    return datetime.now(ZoneInfo("America/Sao_Paulo"))

# --- GERENCIADOR DE TABELAS DIMENSÃO ---
def criar_gerenciador_dimensao(titulo_tabela: str, nome_tabela_db: str):
    with ui.card().classes('w-full p-4 border rounded-lg shadow-sm mb-4'):
        ui.label(titulo_tabela).classes('text-xl font-bold text-gray-800 mb-2')
        res = supabase.table(nome_tabela_db).select("*").order("nome").execute()
        dados = res.data if res.data else []

        columns = [
            {'name': 'nome', 'label': 'Nome', 'field': 'nome', 'align': 'left'},
            {'name': 'id', 'label': '', 'field': 'id', 'align': 'right'}
        ]
        grid = ui.table(columns=columns, rows=dados, row_key='id').classes('w-full mb-4')
        grid.add_slot('body-cell-id', '''
            <q-td :props="props">
                <q-btn icon="edit" size="sm" color="blue" flat @click="$parent.$emit('editar', props.row)" />
                <q-btn icon="delete" size="sm" color="red" flat @click="$parent.$emit('deletar', props.row)" />
            </q-td>
        ''')

        def deletar_item(msg):
            supabase.table(nome_tabela_db).delete().eq('id', msg.args['id']).execute()
            ui.notify('Removido com sucesso!', color='negative')
            ui.navigate.reload()

        def editar_item(msg):
            item = msg.args
            with ui.dialog() as dialog, ui.card().classes('w-80'):
                ui.label(f"Editar {item['nome']}").classes('font-bold text-lg')
                input_edit = ui.input('Novo Nome', value=item['nome']).classes('w-full')
                def salvar():
                    if input_edit.value.strip():
                        supabase.table(nome_tabela_db).update({'nome': input_edit.value.strip()}).eq('id', item['id']).execute()
                        ui.notify("Atualizado com sucesso!", color='positive')
                        dialog.close()
                        ui.navigate.reload()
                ui.button('Salvar', on_click=salvar).classes('w-full bg-blue-700 text-white mt-2')
            dialog.open()

        grid.on('deletar', deletar_item)
        grid.on('editar', editar_item)

        with ui.row().classes('w-full items-center gap-2 mt-2'):
            input_nome = ui.input(f'Novo item em {titulo_tabela}').classes('flex-1')
            def salvar_novo():
                if not input_nome.value.strip():
                    ui.notify('Preencha o nome!', color='warning')
                    return
                supabase.table(nome_tabela_db).insert({"nome": input_nome.value.strip()}).execute()
                ui.notify('Cadastrado com sucesso!', color='positive')
                input_nome.value = ''
                ui.navigate.reload()
                
            ui.button('Adicionar', on_click=salvar_novo, icon='add').classes('bg-green-700 text-white')

        with ui.card().classes('w-full p-4 border rounded-lg shadow-sm bg-blue-50'):
            ui.label('📋 Solicitantes de Acesso Pendentes').classes('text-xl font-bold text-blue-900 mb-2')
            
            solic_res = supabase.table('solicitacoes_acesso').select("*").eq('status', 'PENDENTE').order('created_at', desc=True).execute()
            pedidos = solic_res.data if solic_res.data else []

            if not pedidos:
                ui.label('Nenhuma solicitação pendente no momento.').classes('text-gray-500 italic')
            else:
                for p in pedidos:
                    with ui.card().classes('w-full p-3 bg-white border mb-2'):
                        with ui.row().classes('justify-between items-center w-full'):
                            with ui.column().classes('gap-0'):
                                ui.label(f"✉️ {p['email']}").classes('font-bold text-base')
                                try:
                                    dt_utc = datetime.fromisoformat(p['created_at'].replace('Z', '+00:00'))
                                    dt_br = dt_utc.astimezone(ZoneInfo("America/Sao_Paulo"))
                                    data_fmt = dt_br.strftime("%d/%m/%Y às %H:%M:%S")
                                except:
                                    data_fmt = p['created_at']
                                    
                                ui.label(f"📅 Data/Hora: {data_fmt}").classes('text-xs text-gray-600')
                                ui.label(f"📍 Localização: {p['localizacao']}").classes('text-xs text-gray-600')
                                ui.label(f"💻 Aparelho: {p['dispositivo']}").classes('text-xs text-gray-400')

                            with ui.row().classes('gap-2'):
                                def aprovar(pedido=p):
                                    # Adiciona ou atualiza na tabela de perfis de usuários
                                    supabase.table('perfis_usuarios').upsert({
                                        'email': pedido['email'].lower(),
                                        'senha': pedido['senha_temporaria'],
                                        'ativo': True,
                                        'role': 'USER'
                                    }, on_conflict='email').execute()

                                    # Atualiza solicitação para aprovada
                                    supabase.table('solicitacoes_acesso').update({'status': 'APROVADO'}).eq('id', pedido['id']).execute()
                                    ui.notify(f"Acesso APROVADO para {pedido['email']}!", color='positive')
                                    ui.navigate.reload()

                                def reprovar(pedido=p):
                                    supabase.table('solicitacoes_acesso').update({'status': 'REPROVADO'}).eq('id', pedido['id']).execute()
                                    ui.notify(f"Solicitação de {pedido['email']} reprovada.", color='warning')
                                    ui.navigate.reload()

                                ui.button('Aprovar', on_click=aprovar, icon='check').classes('bg-green-600 text-white text-xs')
                                ui.button('Reprovar', on_click=reprovar, icon='close').classes('bg-red-600 text-white text-xs')

        # --- 2. GESTÃO E CONTROLE DE USUÁRIOS ---
        with ui.card().classes('w-full p-4 border rounded-lg shadow-sm bg-slate-50'):
            ui.label('👥 Controle de Usuários do App').classes('text-xl font-bold text-slate-800 mb-2')
            
            users_res = supabase.table('perfis_usuarios').select("*").order('created_at', desc=True).execute()
            usuarios_lista = users_res.data if users_res.data else []

            if not usuarios_lista:
                ui.label('Nenhum usuário cadastrado.').classes('text-gray-500 italic')
            else:
                for u in usuarios_lista:
                    is_active = u.get('ativo', True)
                    with ui.card().classes(f'w-full p-3 bg-white border mb-2 {"border-l-8 border-l-green-500" if is_active else "border-l-8 border-l-red-500"}'):
                        with ui.row().classes('justify-between items-center w-full'):
                            with ui.column().classes('gap-0'):
                                ui.label(f"👤 {u['email']}").classes('font-bold text-base')
                                status_tag = "🟢 ATIVO" if is_active else "🔴 INATIVO"
                                ui.label(f"Status: {status_tag} | Permissão: {u.get('role', 'USER')}").classes('text-xs text-gray-600')

                            if u['email'] != ADMIN_EMAIL:
                                def alternar_status(user=u):
                                    novo_status = not user.get('ativo', True)
                                    supabase.table('perfis_usuarios').update({'ativo': novo_status}).eq('id', user['id']).execute()
                                    txt_notif = "ativado" if novo_status else "inativado"
                                    ui.notify(f"Usuário {user['email']} {txt_notif} com sucesso!", color='info')
                                    ui.navigate.reload()

                                btn_lbl = 'Inativar' if is_active else 'Ativar'
                                btn_clr = 'bg-red-600' if is_active else 'bg-green-600'
                                ui.button(btn_lbl, on_click=alternar_status).classes(f'{btn_clr} text-white text-xs')

        ui.separator()

        # --- 3. SEÇÃO DAS TABELAS DIMENSÃO ---
        criar_gerenciador_dimensao("1. Categorias de Despesas", "dim_categorias")
        criar_gerenciador_dimensao("2. Formas de Pagamento", "dim_formas_pgto")
